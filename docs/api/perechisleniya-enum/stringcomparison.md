# StringComparison

```csharp
/// <summary>
/// Указывает культуру, регистр и правила сортировки, которые будут использоваться определёнными перегрузками методов System.String.Compare(System.String,System.String) и System.String.Equals(System.Object).
/// </summary>
public enum StringComparison
{
    /// <summary>
    /// Сравнение строк с учетом культурно-зависимых правил сортировки и текущей культуры.
    /// </summary>
    CurrentCulture = 0,
    /// <summary>
    /// Сравнение строк с учетом культурно-зависимых правил сортировки, текущей культуры и игнорированием регистра строк.
    /// </summary> 
    CurrentCultureIgnoreCase = 1,
    /// <summary>
    /// Сравнение строк с учетом культурно-зависимых правил сортировки и инвариантной культуры.
    /// </summary>
    InvariantCulture = 2,
    /// <summary>
    /// Сравнение строк с учетом культурно-зависимых правил сортировки, инвариантной культуры и игнорированием регистра строк.
    /// </summary> 
    InvariantCultureIgnoreCase = 3,
    /// <summary>
    /// Сравнение строк с использованием порядковых (двоичных) правил сортировки.
    /// </summary>
    Ordinal = 4,
    /// <summary>
    /// Сравнение строк с использованием порядковых (двоичных) правил сортировки и игнорированием регистра строк.
    /// </summary>
    OrdinalIgnoreCase = 5
}
```
